Algoritma Konsensus RAFT

Algoritma konsensus RAFT (Replicated And Fault Tolerant) adalah algoritma konsensus yang dirancang untuk menjaga konsistensi data dalam sistem terdistribusi. RAFT menggunakan mekanisme leader election dan log replication untuk memastikan semua node dalam jaringan memiliki salinan data yang identik. Algoritma ini merupakan crash fault tolerant (CFT), yang berarti dapat menoleransi kegagalan node akibat crash, namun tidak dapat menangani node yang berperilaku jahat (Byzantine fault).

Implementasi RAFT pada Hyperledger Fabric v2

Hyperledger Fabric v2 mengimplementasikan algoritma konsensus RAFT sebagai mekanisme default untuk ordering service. Implementasi ini menggunakan Raft consensus type yang memungkinkan jaringan blockchain untuk mencapai konsensus tanpa memerlukan Byzantine fault tolerance, sehingga lebih efisien dalam hal throughput dan latency.

Pseudocode Algoritma RAFT pada Fabric v2

// Algoritma RAFT pada Hyperledger Fabric v2
class RaftNode {
    state: enum {FOLLOWER, CANDIDATE, LEADER}
    currentTerm: int
    votedFor: NodeId | null
    log: Array<LogEntry>
    commitIndex: int
    lastApplied: int
    electionTimeout: Timer
    heartbeatInterval: Timer

    onStartup() {
        state = FOLLOWER
        currentTerm = 0
        votedFor = null
        electionTimeout = random(150ms, 300ms)
        startElectionTimer()
    }

    onElectionTimeout() {
        if (state == FOLLOWER) {
            state = CANDIDATE
            currentTerm++
            votedFor = self
            votesReceived = 1
            
            for node in cluster {
                sendRequestVote(node, currentTerm, 
                    log.lastIndex(), 
                    log.lastTerm())
            }
            
            electionTimeout = random(150ms, 300ms)
            startElectionTimer()
        }
    }

    onReceiveVote(candidateTerm, candidateId, 
                     lastLogIndex, lastLogTerm) {
        if (candidateTerm > currentTerm) {
            currentTerm = candidateTerm
            state = FOLLOWER
            votedFor = null
        }
        
        if (candidateTerm == currentTerm && 
            votedFor == null &&
            isLogUpToDate(lastLogIndex, lastLogTerm)) {
            votedFor = candidateId
            resetElectionTimer()
            return true
        }
        return false
    }

    onReceiveHeartbeat(leaderTerm, leaderId, 
                          prevLogIndex, prevLogTerm,
                          entries, leaderCommit) {
        if (leaderTerm >= currentTerm) {
            state = FOLLOWER
            currentTerm = leaderTerm
            resetElectionTimer()
        }
        
        if (leaderTerm == currentTerm && 
            state != FOLLOWER) {
            state = FOLLOWER
        }
        
        if (matchLog(prevLogIndex, prevLogTerm)) {
            appendEntries(entries)
            commitIndex = min(leaderCommit, 
                log.lastIndex())
            return true
        }
        return false
    }

    onBecomeLeader() {
        state = LEADER
        for node in cluster {
            nextIndex[node] = log.lastIndex() + 1
            matchIndex[node] = 0
        }
        sendHeartbeat()
        heartbeatInterval = setInterval(sendHeartbeat, 50ms)
    }

    onReceiveTransaction(tx) {
        if (state == LEADER) {
            logEntry = new LogEntry(currentTerm, tx)
            log.append(logEntry)
            
            for node in cluster {
                replicateLog(node)
            }
        }
    }
}

Implementasi RAFT pada Hyperledger Fabric v3

Hyperledger Fabric v3 memperkenalkan peningkatan pada implementasi RAFT dengan optimasi performa dan stabilitas yang lebih baik. Versi ini juga menyediakan opsi untuk menggunakan SmartBFT sebagai alternatif konsensus, namun implementasi RAFT standar tetap menjadi pilihan utama untuk skenario yang tidak memerlukan Byzantine fault tolerance.

Pseudocode Algoritma RAFT pada Fabric v3

// Algoritma RAFT pada Hyperledger Fabric v3 (Enhanced)
class RaftNodeV3 {
    state: enum {FOLLOWER, CANDIDATE, LEADER}
    currentTerm: int
    votedFor: NodeId | null
    log: Array<LogEntry>
    commitIndex: int
    lastApplied: int
    electionTimeout: Timer
    heartbeatInterval: Timer
    snapshotIndex: int  // New in v3
    snapshotTerm: int    // New in v3

    onStartup() {
        state = FOLLOWER
        currentTerm = 0
        votedFor = null
        electionTimeout = random(150ms, 300ms)
        snapshotIndex = 0
        snapshotTerm = 0
        loadSnapshot()  // Enhanced: Load from snapshot
        startElectionTimer()
    }

    onElectionTimeout() {
        if (state == FOLLOWER) {
            state = CANDIDATE
            currentTerm++
            votedFor = self
            votesReceived = 1
            
            // Enhanced: Parallel vote requests
            votePromises = parallelRequestVote(cluster, 
                currentTerm, 
                log.lastIndex(), 
                log.lastTerm(),
                snapshotIndex,  // Include snapshot info
                snapshotTerm)
            
            electionTimeout = random(150ms, 300ms)
            startElectionTimer()
        }
    }

    onReceiveVote(candidateTerm, candidateId, 
                     lastLogIndex, lastLogTerm,
                     candidateSnapshotIndex, candidateSnapshotTerm) {
        if (candidateTerm > currentTerm) {
            currentTerm = candidateTerm
            state = FOLLOWER
            votedFor = null
        }
        
        // Enhanced: Check snapshot consistency
        logUpToDate = isLogUpToDate(lastLogIndex, 
            lastLogTerm, 
            candidateSnapshotIndex, 
            candidateSnapshotTerm)
        
        if (candidateTerm == currentTerm && 
            votedFor == null && logUpToDate) {
            votedFor = candidateId
            resetElectionTimer()
            return true
        }
        return false
    }

    onReceiveHeartbeat(leaderTerm, leaderId, 
                          prevLogIndex, prevLogTerm,
                          entries, leaderCommit,
                          leaderSnapshotIndex) {
        if (leaderTerm >= currentTerm) {
            state = FOLLOWER
            currentTerm = leaderTerm
            resetElectionTimer()
        }
        
        // Enhanced: Handle snapshot installation
        if (leaderSnapshotIndex > snapshotIndex) {
            installSnapshot(leaderSnapshotIndex)
        }
        
        if (matchLog(prevLogIndex, prevLogTerm)) {
            appendEntries(entries)
            commitIndex = min(leaderCommit, 
                log.lastIndex())
            
            // Enhanced: Auto-compaction via snapshot
            if (log.size() > snapshotThreshold) {
                createSnapshot()
            }
            
            return true
        }
        return false
    }

    onBecomeLeader() {
        state = LEADER
        for node in cluster {
            nextIndex[node] = log.lastIndex() + 1
            matchIndex[node] = 0
        }
        sendHeartbeat()
        // Enhanced: Optimized heartbeat interval
        heartbeatInterval = setInterval(sendHeartbeat, 40ms)
    }

    onReceiveTransaction(tx) {
        if (state == LEADER) {
            logEntry = new LogEntry(currentTerm, tx)
            log.append(logEntry)
            
            // Enhanced: Batch replication for better throughput
            batchSize = calculateOptimalBatchSize()
            for node in cluster {
                replicateLogBatch(node, batchSize)
            }
        }
    }

    createSnapshot() {
        // Enhanced: Create snapshot for log compaction
        snapshot = captureState(commitIndex)
        snapshotIndex = commitIndex
        snapshotTerm = log[commitIndex].term
        saveSnapshot(snapshot)
        compactLog(snapshotIndex)
    }
}
